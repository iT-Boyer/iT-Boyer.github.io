func factorial(n: Int) -> Int {
    if n <= 1 { return n }
    return n * factorial(n: n - 1)
}

let number = 4
print("\(number)! is equal to \(factorial(n: number))")
  1. swiftc命令 运行swiftc命令-g选项生成swift调试信息,在目录中生存可执行的Factorial文件：
